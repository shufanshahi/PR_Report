\begin{frame}{Motivation}
  \begin{itemize}
    \setlength\itemsep{5mm}
    \item \textbf{\color{Navy}Distribution shift:} test data differs from training data
    \item Accuracy drops sharply. Retraining is usually impossible.
    \item \textbf{\color{Navy}Test-Time Adaptation:} adapt a pretrained model online, using only the unlabeled test stream
    \item No source data. No target labels. Model access only.
  \end{itemize}
  \vspace{6mm}
  \begin{card}
    \centering\sffamily\color{Navy}
    Open-world TTA: stay accurate on known classes, and reject unknown ones.
  \end{card}
\end{frame}

  \sectiontag{The gap}
  \vspace{3mm}
  \begin{itemize}
    \setlength\itemsep{2.5mm}
    \item Most TTA assumes a \textbf{closed world}
    \item Real streams contain \textbf{unknown-class outliers}
    \item Joint recognition and rejection: under-explored
  \end{itemize}
\end{frame}

  \begin{card}

\section{This Work}
\begin{frame}{STAMP: The Pipeline}
  \begin{columns}[T]
    \begin{column}{0.185\textwidth}\begin{cardh}{2.8cm}\cardcol{clone}{Steel}{Augment}{15 views}\end{cardh}\end{column}
    \begin{column}{0.014\textwidth}\vspace{12mm}\centering{\color{Steel}$\to$}\end{column}
    \begin{column}{0.185\textwidth}\begin{cardh}{2.8cm}\cardcol{check-double}{Steel}{Consistency}{filter}\end{cardh}\end{column}
    \begin{column}{0.014\textwidth}\vspace{12mm}\centering{\color{Steel}$\to$}\end{column}
    \begin{column}{0.185\textwidth}\begin{cardh}{2.8cm}\cardcol{filter}{Orange}{Entropy Gate}{$H(x)<m$}\end{cardh}\end{column}
    \begin{column}{0.014\textwidth}\vspace{12mm}\centering{\color{Steel}$\to$}\end{column}
    \begin{column}{0.185\textwidth}\begin{cardh}{2.8cm}\cardcol{database}{Steel}{Memory}{capacity 64}\end{cardh}\end{column}
    \begin{column}{0.014\textwidth}\vspace{12mm}\centering{\color{Steel}$\to$}\end{column}
    \begin{column}{0.185\textwidth}\begin{cardh}{2.8cm}\cardcol{redo}{Orange}{Replay}{weighted update}\end{cardh}\end{column}
  \end{columns}
  \vspace{3mm}
  \small
  \[
    \mathcal{L}_{\mathrm{STAMP}} = \frac{1}{|\mathcal{M}|}\sum_{x_i\in\mathcal{M}} \widetilde{w}_i\, H\!\left(f_\theta(x_i)\right),
    \qquad
    \widetilde{w}_i=\frac{|\mathcal{M}|\,w_i}{\sum_j w_j}
  \]
  \vspace{2mm}
  \begin{itemize}
    \setlength\itemsep{1.5mm}\footnotesize
    \item Updates only normalization parameters
    \item \textbf{The catch:} entropy decides admission \emph{and} the loss
  \end{itemize}
\end{frame}

% =========================================================
\begin{frame}{Setup}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \setlength\itemsep{3mm}
        \item \textbf{\color{Navy}ID samples:} classify correctly
        \item \textbf{\color{Navy}OOD samples:} reject, never adapt on them
        \item No test labels used, ever
        \item CIFAR-10-C / 100-C, severity 5, 15 corruptions
        \item ResNet-18, batch 64
      \end{itemize}
    \end{column}
    \begin{column}{0.44\textwidth}
      \textbf{\color{Navy}Metrics:} Accuracy, AUC, and
      \[
        H\text{-score} = \frac{2\,\mathrm{Acc}\cdot\mathrm{AUC}}{\mathrm{Acc}+\mathrm{AUC}}
      \]
      \vspace{2mm}
      \begin{cardh}{1.8cm}
        \centering\sffamily\color{Navy}\small
        Change only the score. Hold the rest of STAMP fixed.
      \end{cardh}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Roadmap}
  \begin{columns}[T]

  \end{columns}
  \vspace{6mm}
  \begin{itemize}
    \setlength\itemsep{3mm}
    \item R1, R2: change \textbf{what} scores a sample
    \item R3: change \textbf{how} the score becomes a decision
    \item R4: fixes a bug in the replay loss
  \end{itemize}
\end{frame}

% =========================================================
\begin{frame}{Round 1: KNN Feature Distance}
  \[
    s_{\mathrm{KNN}}(x) = 1 - \frac{1}{k}\sum_{j\in\mathcal{N}_k(x)} \hat{\mathbf{h}}(x)^{\top}\hat{\mathbf{r}}_j
  \]
  \begin{itemize}
    \setlength\itemsep{2.5mm}
    \item Distance to a bank of clean source features
    \item $k{=}10$, up to 2{,}000 references
  \end{itemize}
  \vspace{5mm}
  \sectiontag{Result}
  \vspace{2mm}
  \begin{itemize}
    \setlength\itemsep{2mm}
    \item Accuracy up: \textbf{\color{Steel}+21.3 pts}
    \item AUC down: \textbf{\color{Orange}$-$9.4 / $-$11.0 pts}
    \item Cause: corruption moves the reference bank away from clean ID features
  \end{itemize}
\end{frame}

% =========================================================
\begin{frame}{Round 2: Logit Energy Score}
  \[
    E_T(x) = -T\log\sum_{c=1}^{C}\exp(z_c(x)/T)
  \]
  \begin{itemize}
    \setlength\itemsep{2.5mm}
    \item No reference bank, no extra forward pass
    \item Threshold: 80th percentile of the last 1{,}000 scores
  \end{itemize}
  \vspace{5mm}
  \sectiontag{Result}
  \vspace{2mm}
  \begin{itemize}
    \setlength\itemsep{2mm}
    \item CIFAR-100-C: strongest result yet (\textbf{\color{Steel}71.33 H})
    \item CIFAR-10-C: AUC drops to \textbf{\color{Orange}54.33}
    \item Not dataset-independent
  \end{itemize}
\end{frame}

% =========================================================
\begin{frame}{Round 3: Adaptive Threshold}
  \[
    \alpha_{t+1} = \operatorname{clip}_{[0,1]}\big(\alpha_t + \gamma(\rho - r_t)\big)
  \]
  \begin{itemize}
    \setlength\itemsep{2.5mm}
    \item Target rejection rate $\rho = 0.20$
    \item Cutoff adjusts automatically from the rolling score buffer
  \end{itemize}
  \vspace{5mm}
  \sectiontag{Result}
  \vspace{2mm}
  \begin{itemize}
    \item Matches, but does not improve, the fixed-threshold energy score
  \end{itemize}
\end{frame}

% =========================================================
\begin{frame}{Round 4: The Bug}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{cardh}{2.3cm}
      \centering{\sffamily\bfseries\color{Orange}\faIcon{bug}\ \ Before}\\[2mm]
      \[
        \mathcal{L}_{\mathrm{old}} = \frac{64}{N}\,\mathcal{L}
      \]
      \end{cardh}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{cardh}{2.3cm}
      \centering{\sffamily\bfseries\color{Steel}\faIcon{check-circle}\ \ After}\\[2mm]
      \[
        \mathcal{L} = \frac{1}{N}\sum_i \widetilde{w}_i H_i
      \]
      \end{cardh}
    \end{column}
  \end{columns}
  \vspace{5mm}
  \begin{itemize}
    \setlength\itemsep{2.5mm}
    \item Memory starts partially filled ($N{<}64$)
    \item Gradients were inflated up to \textbf{64$\times$}
    \item Fixed variant: \textbf{STAMP+BSN}
  \end{itemize}
  \vspace{3mm}
  \sectiontag{Result}
  \vspace{2mm}
  \begin{itemize}
    \item Best on both benchmarks: \textbf{\color{Steel}81.19 H} (C10), \textbf{\color{Steel}72.89 H} (C100)
  \end{itemize}
\end{frame}

\begin{frame}{Key Takeaways}
  \begin{itemize}
    \setlength\itemsep{5.5mm}
    \item[\iconcircle{crosshairs}{Steel}] \textbf{\color{Navy}KNN:} better accuracy, worse detection
    \item[\iconcircle{bolt}{Orange}] \textbf{\color{Navy}Energy:} strong on CIFAR-100-C, dataset-dependent
    \item[\iconcircle{bug}{Orange}] \textbf{\color{Navy}Loss fix:} best on both benchmarks
    \item[\iconcircle{sliders-h}{Steel}] \textbf{\color{Navy}Calibration:} no improvement over a fixed threshold
  \end{itemize}
\end{frame}

% =========================================================
\begin{frame}{Conclusion}
  \begin{itemize}
    \setlength\itemsep{4.5mm}
    \item No single score is dataset-independent
    \item The biggest gain came from a \textbf{correctness fix}, not a new score
    \item STAMP+BSN: best H-score on both benchmarks
  \end{itemize}
\end{frame}
