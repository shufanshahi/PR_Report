\begin{frame}{Motivation}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{itemize}
        \setlength\itemsep{3.5mm}
        \item Deployment data rarely matches training data
        \item[] \textbf{\color{Navy}Distribution shift} --- mismatch between source and target distributions
        \item Accuracy degrades sharply once shift appears
        \item Retraining is often impossible: no source data, no labels, must react instantly
      \end{itemize}

% =========================================================
\begin{frame}{What Is Test-Time Adaptation?}
  \begin{quote}
    \itshape\color{Navy}``Adapting a pretrained model to unlabeled test data --- on the fly, at deployment.''
  \end{quote}
  \vspace{1mm}
  TTA updates a source-trained model using only the unlabeled target stream, guided by self-supervised signals (entropy, pseudo-labels, feature statistics) --- no retraining, no target labels.
  \vspace{3mm}

  \sectiontag{Core assumptions}
  \vspace{2mm}
  \begin{columns}[T]

\begin{frame}{Challenges $\to$ Research Gap}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \sectiontag{Known challenges}\\[2mm]
      \begin{itemize}
        \setlength\itemsep{2.6mm}
        \item \textbf{\color{Navy}Catastrophic forgetting} under continual updates
        \item \textbf{\color{Navy}Confirmation bias} from noisy pseudo-labels
        \item \textbf{\color{Navy}Unstable optimization} under severe shift
        \item \textbf{\color{Navy}Warm-up statistics} are unreliable
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \sectiontag{The open-world gap}\\[2mm]
      \begin{itemize}
        \setlength\itemsep{2.6mm}
        \item Most TTA assumes a \emph{closed world}: every sample is a known class
        \item Real streams contain \textbf{\color{Navy}unknown-class outliers}
        \item Adapting on confident errors \textbf{reinforces mistakes}
        \item Joint recognition + rejection is \textbf{under-explored}
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{4mm}
  \begin{card}
    \centering\sffamily\color{Navy} Our target: stay accurate \emph{and} learn to reject what the model has never seen.
  \end{card}
\end{frame}

\begin{frame}{Our Chosen Direction}
  \begin{cardh}{2.7cm}
  \centering
  \iconcircle{compass}{Orange}\\[2mm]
  {\sffamily\color{Navy}\large Open-world TTA: stay accurate \textbf{without} letting unknown samples corrupt the update.}
  \end{cardh}
  \vspace{5mm}
  \begin{itemize}
    \setlength\itemsep{4mm}
    \item We adopt \textbf{\color{Navy}STAMP} (Yu et al., ECCV 2024) --- built for exactly this joint setting
    \item Its detector \emph{and} its adaptation objective both rely on \textbf{prediction entropy} --- one signal, two jobs
    \item \textbf{Research question:} can the detector be replaced by a signal from OOD detection, leaving the rest of STAMP untouched?
  \end{itemize}
\end{frame}

\section{This Work}
\begin{frame}{STAMP: The Pipeline We Build On}
  \begin{columns}[T]
    \begin{column}{0.185\textwidth}\begin{cardh}{3.0cm}\cardcol{clone}{Steel}{Augment}{15 views, averaged}\end{cardh}\end{column}
    \begin{column}{0.014\textwidth}\vspace{13mm}\centering{\color{Steel}$\to$}\end{column}
    \begin{column}{0.185\textwidth}\begin{cardh}{3.0cm}\cardcol{check-double}{Steel}{Consistency}{drop disagreeing samples}\end{cardh}\end{column}
    \begin{column}{0.014\textwidth}\vspace{13mm}\centering{\color{Steel}$\to$}\end{column}
    \begin{column}{0.185\textwidth}\begin{cardh}{3.0cm}\cardcol{filter}{Orange}{Entropy Gate}{$H(x)<m=\alpha\log C$}\end{cardh}\end{column}
    \begin{column}{0.014\textwidth}\vspace{13mm}\centering{\color{Steel}$\to$}\end{column}
    \begin{column}{0.185\textwidth}\begin{cardh}{3.0cm}\cardcol{database}{Steel}{Balanced Memory}{capacity 64, class-balanced}\end{cardh}\end{column}
    \begin{column}{0.014\textwidth}\vspace{13mm}\centering{\color{Steel}$\to$}\end{column}
    \begin{column}{0.185\textwidth}\begin{cardh}{3.0cm}\cardcol{redo}{Orange}{Weighted Replay}{confidence-weighted update}\end{cardh}\end{column}
  \end{columns}
  \small
  \setlength{\abovedisplayskip}{2pt}\setlength{\belowdisplayskip}{2pt}
  \[
    \mathcal{L}_{\mathrm{STAMP}} = \frac{1}{|\mathcal{M}|}\sum_{x_i\in\mathcal{M}} \widetilde{w}_i\, H\!\left(f_\theta(x_i)\right),
    \qquad
    \widetilde{w}_i=\frac{|\mathcal{M}|\,w_i}{\sum_j w_j},\ \ w_i = \exp[-H(f_\theta(x_i))]
  \]
  \footnotesize
  $\Rightarrow$ Only normalization parameters are updated, via the SAM-style two-step update. \textbf{The catch:} entropy $H(x)$ decides \emph{both} admission \emph{and} the loss --- coupling detection to a classifier whose confidence keeps changing.
\end{frame}

% =========================================================
\begin{frame}{Problem Formulation \& Experimental Principle}
  Let $f_\theta$ be a source-trained classifier with logits $\mathbf{z}(x) = f_\theta(x)$. The test stream mixes:
  \begin{itemize}
    \setlength\itemsep{1.5mm}
    \item corrupted \textbf{in-distribution (ID)} samples --- must still be classified correctly
    \item \textbf{out-of-distribution (OOD)} samples --- scored as outliers, never used as adaptation targets
  \end{itemize}
  \vspace{2mm}
  No test label is ever used for adaptation or threshold fitting --- only for evaluation.
  \vspace{5mm}
  \begin{cardh}{2.6cm}
    \centering\sffamily\color{Navy}
    \textbf{Guiding principle:} change only the outlier score, hold the rest of STAMP fixed. \\
    \footnotesize\color{SubGray} Any performance change is then attributable to the score alone.
  \end{cardh}
\end{frame}

% =========================================================
\begin{frame}{Evaluation Protocol}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \setlength\itemsep{3mm}
        \item \textbf{\color{Navy}Datasets:} CIFAR-10-C, CIFAR-100-C --- severity 5, 15 corruptions
        \item \textbf{\color{Navy}OOD mixture:} 10{,}000 ID + 2{,}500 noise outliers per corruption (80:20)
        \item \textbf{\color{Navy}Backbone:} ResNet-18, batch 64, reset every corruption domain
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \setlength\itemsep{3mm}
        \item \textbf{\color{Navy}Baselines:} source model, \texttt{norm\_test}, modified STAMP
        \item \textbf{\color{Navy}Metrics:} Accuracy (ID), AUC (ID vs.\ OOD), and the joint
      \end{itemize}
      \[
        H\text{-score} = \frac{2\,\mathrm{Acc}\cdot\mathrm{AUC}}{\mathrm{Acc}+\mathrm{AUC}}
      \]
    \end{column}
  \end{columns}
\end{frame}

  \vspace{5mm}
  \begin{itemize}
    \setlength\itemsep{2.5mm}
    \item R1--R2 replace \textbf{\emph{what}} scores a sample; R3 replaces \textbf{\emph{how}} it becomes a decision
    \item R4 is not a new score --- it's a correctness fix to STAMP's own replay loss
    \item Every round keeps the consistency filter, memory, and replay objective fixed
  \end{itemize}

\begin{frame}{Round 1 --- KNN Feature-Distance Score}
  \textbf{\color{Navy}Hypothesis:} feature geometry catches outliers an entropy threshold misses, decoupled from the classifier's own confidence.
  \vspace{3mm}
  \[
    s_{\mathrm{KNN}}(x) = 1 - \frac{1}{k}\sum_{j\in\mathcal{N}_k(x)} \hat{\mathbf{h}}(x)^{\top}\hat{\mathbf{r}}_j
  \]
  \begin{itemize}
    \setlength\itemsep{2.5mm}
    \item $\hat{\mathbf{h}}(x)$: normalized penultimate feature; $\mathcal{R}$: bank of clean source features
    \item Up to 2{,}000 references, $k{=}10$; replaces entropy for \emph{both} admission and AUC
    \item Consistency gate, memory, and replay update stay unchanged
  \end{itemize}
  \vspace{4mm}
  \begin{cardh}{1.8cm}\small\color{Navy}
    Expected: independence from softmax confidence $\Rightarrow$ better AUC. \textbf{Not what we observed.}
  \end{cardh}
\end{frame}

\begin{frame}{Round 2 --- Energy Score with Online Calibration}
  \textbf{\color{Navy}Hypothesis:} drop the fixed reference bank; score directly from logits.
  \vspace{1mm}
  \[
    E_T(x) = -T\log\sum_{c=1}^{C}\exp(z_c(x)/T), \qquad T{=}1
  \]
  \begin{itemize}
    \setlength\itemsep{2.5mm}
    \item Uses logits STAMP already computes --- \textbf{no extra forward pass, no reference bank}
    \item Threshold = \textbf{80th percentile} of the latest 1{,}000 scores (matches 80:20 mixture, label-free)
    \item First 256 samples fall back to entropy margin (warm-up); cutoff resets each domain
  \end{itemize}
  \vspace{1mm}
  \[
    E_T(x) < \operatorname{Quantile}_{0.80}(\mathcal{B}_t)
  \]
\end{frame}

\begin{frame}{Round 3 --- Adaptive Conformal Thresholding}
  Rounds 1--2 change the \emph{score}; Round 3 asks: does \textbf{online threshold calibration} close the gap, independent of score choice?
  \vspace{3mm}
  \[
    \alpha_{t+1} = \operatorname{clip}_{[0,1]}\big(\alpha_t + \gamma(\rho - r_t)\big)
  \]
  \begin{itemize}
    \setlength\itemsep{2.5mm}
    \item $\rho = 0.20$: target rejection rate; $r_t$: observed rate in the current batch
    \item Cutoff read at quantile $1-\alpha_t$ of the rolling buffer --- rises/falls automatically
  \end{itemize}
  \vspace{2mm}
  \begin{cardh}{1.7cm}
    \centering\small\textbf{\color{Navy}Energy--STAMP + ACI}\\
    {\footnotesize Holds the energy score fixed and tests whether adaptive calibration improves admission.}
  \end{cardh}
  \vspace{3mm}
  \vspace{2mm}
  \centerline{\footnotesize\itshape\color{SubGray}Energy+ACI closely matches, but does not improve on, the fixed-percentile energy baseline.}
\end{frame}

\begin{frame}{Round 4 --- The Bug: Replay-Loss Scaling}
  Not a new score --- an \textbf{implementation fix} in STAMP's own replay loss.
  \vspace{2mm}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{cardh}{3.0cm}
      \centering{\sffamily\bfseries\color{Orange}\faIcon{bug}\ \ Before (bug)}\\[2mm]
      \[
        \mathcal{L}_{\mathrm{old}} = \frac{64}{N}\,\mathcal{L}
      \]
      {\footnotesize\color{SubGray} Normalized for capacity 64, then averaged over the \emph{actual} size $N$}
      \end{cardh}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{cardh}{3.0cm}
      \centering{\sffamily\bfseries\color{Steel}\faIcon{check-circle}\ \ After (fixed)}\\[2mm]
      \[
        \widetilde{w}_i = \frac{N w_i}{\sum_j w_j}, \quad \mathcal{L} = \frac{1}{N}\sum_i \widetilde{w}_i H_i
      \]
      {\footnotesize\color{SubGray} Coefficients always sum to $N$ --- a proper weighted average}
      \end{cardh}
    \end{column}
  \end{columns}
  \vspace{1.5mm}
  \footnotesize
  \begin{itemize}
    \setlength\itemsep{0.6mm}
    \item Memory resets every corruption domain and starts \textbf{partially filled} ($N<64$) $\Rightarrow$ gradients inflated up to $\mathbf{64\times}$ right after every reset
    \item Full-memory case ($N{=}64$) is unchanged --- only warm-up is rescaled. We name this fix \textbf{STAMP+BSN}.
  \end{itemize}
\end{frame}

% =========================================================
\begin{frame}{Key Takeaways}
  \begin{itemize}
    \setlength\itemsep{5mm}
    \item[\iconcircle{crosshairs}{Steel}] \textbf{\color{Navy}KNN distance} improves accuracy but \textbf{worsens} detection --- the clean reference bank is the wrong anchor once ID features drift.
    \item[\iconcircle{bolt}{Orange}] \textbf{\color{Navy}Energy scoring} excels on CIFAR-100-C (71.33 H) but is \textbf{dataset-dependent} --- weak on CIFAR-10-C.
    \item[\iconcircle{bug}{Orange}] \textbf{\color{Navy}The replay-loss fix (R4)} gives the \textbf{best results on both benchmarks} --- a bug fix beat every scoring idea.
    \item[\iconcircle{sliders-h}{Steel}] \textbf{Adaptive calibration (R3)} closely matches the fixed-percentile energy baseline but does not improve it.
  \end{itemize}
\end{frame}

% =========================================================
\begin{frame}{Conclusion}
  We asked whether STAMP's entropy-based detector --- which doubles as its adaptation objective --- can be replaced by an OOD-detection signal, holding the rest of STAMP fixed.
  \vspace{3mm}
  \begin{itemize}
    \setlength\itemsep{3.5mm}
    \item Neither KNN distance nor energy scoring alone gives a \textbf{dataset-independent} gain in the joint trade-off
    \item The biggest, most consistent gain came from a \textbf{correctness fix} (STAMP+BSN) --- 81.19 H on CIFAR-10-C, 72.89 on CIFAR-100-C, best on both
    \item ACI threshold calibration does not improve the fixed-percentile energy baseline on either benchmark
  \end{itemize}
\end{frame}
